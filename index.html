function doGet(e) { var action = e.parameter.action; var email = e.parameter.email ? e.parameter.email.toLowerCase().trim() : ""; var userOtp = e.parameter.otp ? e.parameter.otp.trim() : ""; var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var cache = CacheService.getScriptCache(); // 1. ACTION: SEND OTP VIA EMAIL if (action === "sendOtp") { if (!email) { return responseJSON({ status: "error", message: "Email required" }); } // Generate 6-Digit Random OTP var generatedOtp = Math.floor(100000 + Math.random() * 900000).toString(); // Store OTP in cache for 10 minutes cache.put(email + "_otp", generatedOtp, 600); // Send OTP email using GmailApp var subject = "Your BSN Portal Access OTP Code"; var body = "Hello,\n\nYour 6-digit OTP code to access the BSN Academic Portal is: " + generatedOtp + "\n\nThis code will expire in 10 minutes.\n\nRegards,\nMudassir Hussain Khan"; try { GmailApp.sendEmail(email, subject, body); return responseJSON({ status: "success", message: "OTP sent successfully" }); } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } // 2. ACTION: VERIFY OTP if (action === "verifyOtp") { var cachedOtp = cache.get(email + "_otp"); if (cachedOtp && cachedOtp === userOtp) { // Direct access check from cells A2 (Status) and B2 (Access) var status = sheet.getRange("A2").getValue().toString().trim().toLowerCase(); var access = sheet.getRange("B2").getValue().toString().trim().toLowerCase(); return responseJSON({ status: status, // "approved" or "pending" access: access, // "all" or "pediatric, reproductive" email: email }); } else { return responseJSON({ status: "invalid_otp", message: "Incorrect or expired OTP" }); } } return responseJSON({ status: "error", message: "Invalid action" }); } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
